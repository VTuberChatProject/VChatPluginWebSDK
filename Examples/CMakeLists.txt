# 从开发契约生成 C++ 消息并构建 WebSocket 示例。
cmake_minimum_required(VERSION 3.24)
project(vchat_web_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
protobuf_generate_cpp(PROTO_SOURCES PROTO_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../external_plugin.proto")
add_executable(vchat_web_client sdk-client.cpp ${PROTO_SOURCES} ${PROTO_HEADERS})
target_include_directories(vchat_web_client PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(vchat_web_client PRIVATE protobuf::libprotobuf Boost::system)
if(WIN32)
  target_link_libraries(vchat_web_client PRIVATE ws2_32 mswsock)
endif()
if(MSVC)
  target_compile_options(vchat_web_client PRIVATE /utf-8)
endif()
